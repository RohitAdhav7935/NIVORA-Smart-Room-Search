let allRooms = [];


// =================================
// LOAD DATA
// =================================

google.script.run

  .withSuccessHandler(function(data) {

    allRooms = data || [];

    createFilters();

    displayRooms(allRooms);

  })

  .withFailureHandler(function(error) {

    document.getElementById("roomGrid").innerHTML =

      `<div class="no-results">
        ❌ Error loading data.<br><br>
        ${error.message}
      </div>`;

  })

  .getRooms();


// =================================
// CREATE FILTERS
// =================================

function createFilters() {

  const propertyTypes = new Set();

  const furnishing = new Set();

  const suitable = new Set();


  allRooms.forEach(room => {

    if (room.propertyType)
      propertyTypes.add(room.propertyType);

    if (room.furnishing)
      furnishing.add(room.furnishing);

    if (room.suitableFor)
      suitable.add(room.suitableFor);

  });


  fillSelect(
    "propertyFilter",
    propertyTypes
  );

  fillSelect(
    "furnishingFilter",
    furnishing
  );

  fillSelect(
    "suitableFilter",
    suitable
  );

}


// =================================
// FILL SELECT
// =================================

function fillSelect(id, values) {

  const select =
    document.getElementById(id);


  [...values]
    .sort()
    .forEach(value => {

      const option =
        document.createElement("option");

      option.value = value;

      option.textContent = value;

      select.appendChild(option);

    });

}


// =================================
// FILTER ROOMS
// =================================

function filterRooms() {

  const search =
    document
      .getElementById("searchInput")
      .value
      .toLowerCase()
      .trim();


  const property =
    document
      .getElementById("propertyFilter")
      .value;


  const furnishing =
    document
      .getElementById("furnishingFilter")
      .value;


  const suitable =
    document
      .getElementById("suitableFilter")
      .value;


  const filtered =
    allRooms.filter(room => {

      const searchable = [

        room.area,

        room.address,

        room.propertyType,

        room.furnishing,

        room.suitableFor,

        room.rent

      ]

      .join(" ")
      .toLowerCase();


      const matchesSearch =
        !search ||
        searchable.includes(search);


      const matchesProperty =
        !property ||
        room.propertyType === property;


      const matchesFurnishing =
        !furnishing ||
        room.furnishing === furnishing;


      const matchesSuitable =
        !suitable ||
        room.suitableFor === suitable;


      return (

        matchesSearch &&
        matchesProperty &&
        matchesFurnishing &&
        matchesSuitable

      );

    });


  displayRooms(filtered);

}


// =================================
// DISPLAY ROOMS
// =================================

function displayRooms(rooms) {

  const grid =
    document.getElementById("roomGrid");


  document.getElementById("resultCount")
    .textContent =
      rooms.length +
      " room/property listing(s) found";


  if (!rooms.length) {

    grid.innerHTML = `

      <div class="no-results">

        <div style="font-size:45px;">
          🏠
        </div>

        <h3>
          No rooms found
        </h3>

        <p>
          Try changing your search or filters.
        </p>

      </div>

    `;

    return;

  }


  grid.innerHTML = "";


  rooms.forEach(room => {

    grid.appendChild(
      createRoomCard(room)
    );

  });

}

// =================================
// DETAIL HTML
// =================================

function detail(label, value) {

  if (!value) {

    return "";

  }


  return `

    <div class="detail">

      <strong>
        ${label}
      </strong>

      ${safe(value)}

    </div>

  `;

}


// =================================
// PHOTO URL
// =================================

function extractPhotoURL(value) {

  if (!value) return "";


  const text =
    String(value).trim();


  // Google Drive URL
  const driveMatch =
    text.match(
      /https?:\/\/drive\.google\.com\/[^\s,]+/i
    );


  if (driveMatch) {

    return convertDriveURL(
      driveMatch[0]
    );

  }


  // Normal image URL
  const urlMatch =
    text.match(
      /https?:\/\/[^\s,]+/i
    );


  return urlMatch
    ? urlMatch[0]
    : "";

}


// =================================
// GOOGLE DRIVE IMAGE URL
// =================================

function convertDriveURL(url) {

  const match =
    url.match(
      /[-\w]{25,}/
    );


  if (!match) {

    return url;

  }


  return (
    "https://drive.google.com/thumbnail?id=" +
    match[0] +
    "&sz=w1000"
  );

}


// =================================
// SAFE TEXT
// =================================

function safe(value) {

  return String(value || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");

}
